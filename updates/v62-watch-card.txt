function SignalCard({signal}:{signal:RadarSignal}){
 const confirmed=signal.phase==='CONFIRMED',up=signal.direction==='LONG';
 const stale=signal.watchState==='STALE',held=signal.watchState==='HELD';
 return <div className={'rounded-2xl border p-3.5 space-y-2.5 '+(stale?'border-slate-600/30 bg-[#18212d]':confirmed?'border-cyan-400/30 bg-gradient-to-br from-[#153b47] to-[#18223c]':'border-amber-400/25 bg-gradient-to-br from-[#35291e] to-[#1d2437]')}>
  <div className="flex items-start justify-between gap-2">
   <div className="min-w-0"><div className="flex items-center gap-2"><strong className="text-sm text-white">{pairName(signal.symbol)}</strong><span className="rounded-md border border-white/15 px-1.5 py-0.5 text-[10px] text-slate-300">{(signal.frame||'5m').toUpperCase()}</span></div>
    <span className={'mt-1.5 block text-[11px] font-semibold '+(stale?'text-slate-400':confirmed?'text-cyan-200':'text-amber-200')}>{stale?'Нет свежих данных':confirmed?'Подтверждённое наблюдение':held?'Под наблюдением':'Подготовка WATCH'}</span>
   </div>
   <span className={'shrink-0 flex gap-1 items-center text-xs font-bold '+(up?'text-emerald-300':'text-rose-300')}>{up?<ArrowUpRight size={16}/>:<ArrowDownRight size={16}/>} {signal.direction}</span>
  </div>
  <p className="text-xs leading-relaxed text-slate-300">{signal.reason}</p>
  {signal.phase==='WATCH'&&signal.watchNote&&<p className="text-[11px] text-amber-200/90">{signal.watchNote}</p>}
  <div className="grid grid-cols-2 gap-2 border-t border-white/10 pt-2 text-[11px]">
   <span className="text-slate-400">Уровень <strong className="block font-mono text-slate-200">{fmt(signal.level)}</strong></span>
   <span className="text-slate-400 text-right">Цена <strong className="block font-mono text-slate-200">{fmt(signal.price)}</strong></span>
  </div>
  <p className="text-[10px] text-slate-500">Снимок {stamp(signal.timestamp)} · Нажми для графика и подробностей</p>
 </div>;
}
