 const bybitAssistantTargets=useMemo(()=>[...new Set(assistantLedger.orders
  .filter(o=>o.exchange==='bybit'&&(o.state==='WAITING'||o.state==='OPEN'))
  .map(o=>o.symbol+':'+o.frame))].sort().slice(0,6),[assistantLedger]);
 const bybitTargetKey=bybitAssistantTargets.join('|');
 useEffect(()=>{
  if(!bybitTargetKey)return;
  let mounted=true,inFlight=false;
  const check=async()=>{
   if(document.hidden||inFlight)return;
   inFlight=true;
   try{
    const targets=bybitTargetKey.split('|');
    const result=await Promise.allSettled(targets.map(async value=>{
     const separator=value.lastIndexOf(':');
     const symbol=value.slice(0,separator),frame=value.slice(separator+1) as '5m'|'1h'|'4h';
     const bars=await fetchBybitSpotClosedBars(symbol,frame,Date.now());
     return {key:'bybit:'+value,bars};
    }));
    if(!mounted)return;
    const complete=result.filter((x):x is PromiseFulfilledResult<{key:string;bars:DemoBar[]}>=>x.status==='fulfilled')
     .map(x=>x.value);
    if(complete.length)setBybitAssistantBars(old=>{
     const next={...old};
     for(const e of complete)next[e.key]=e.bars;
     return next;
    });
    const failed=result.find((x):x is PromiseRejectedResult=>x.status==='rejected');
    setBybitAssistantError(failed?String(failed.reason):'');
   }finally{inFlight=false;}
  };
  void check();
  const interval=setInterval(()=>{void check();},60000);
  const resume=()=>{if(!document.hidden)void check();};
  document.addEventListener('visibilitychange',resume);
  return ()=>{mounted=false;clearInterval(interval);document.removeEventListener('visibilitychange',resume);};
 },[bybitTargetKey]);
 useEffect(()=>{
  if(!assistantLedger.orders.some(o=>o.state==='WAITING'||o.state==='OPEN'))return;
  const closedBars:Record<string,DemoBar[]>={...bybitAssistantBars};
  const reversals:Record<string,ObservedReversal>={};
  for(const o of assistantLedger.orders){
   if(o.state!=='WAITING'&&o.state!=='OPEN')continue;
   const asset=assets[o.symbol]||cache.current[o.symbol];
   if(o.exchange==='binance'){
    const raw=asset?.frames[o.frame]||(o.frame==='5m'?asset?.candles:undefined)||[];
    closedBars['binance:'+o.symbol+':'+o.frame]=raw
     .filter(c=>c.closed!==false).map(c=>({time:c.time,open:c.open,high:c.high,low:c.low,close:c.close,closed:true}));
   }
   const sig=asset?.signals[o.frame];
   if(sig)reversals[o.symbol+':'+o.frame]={direction:sig.direction,phase:sig.phase,timestamp:sig.timestamp};
  }
  setAssistantLedger(prev=>updateAssistantLedger(prev,scenarios,closedBars,reversals,Date.now()));
 },[assistantLedger,scenarios,assets,bybitAssistantBars,now]);
