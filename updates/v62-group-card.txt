function GroupCard({group,now,onDemo,savedIds,asset}:{group:ReturnType<typeof groupedOpen>[number];now:number;onDemo?:(s:Scenario)=>void;savedIds?:Set<string>;asset?:AssetState}){
 const [expanded,setExpanded]=useState(false);
 const lead=group.scenarios[0];
 const passport=evaluatePassport(lead,asset?{price:asset.price,lastTick:asset.lastTick,signals:asset.signals,volumeRatio:asset.volumeRatio}:null,now);
 return <section className="apex-aura-signals rounded-2xl p-3 space-y-2.5">
  <div className="flex items-center justify-between gap-2">
   <div className="min-w-0"><strong className="block text-base text-slate-100">{pairName(lead.symbol)}</strong>
    <span className="block text-[11px] text-violet-200 mt-1">{group.scenarios.map(s=>(s.frame||'5m').toUpperCase()).join(' + ')} · {lead.direction}</span>
   </div>
   <span className={'shrink-0 rounded-lg px-2.5 py-1 text-[11px] font-medium '+(group.conflict?'bg-amber-400/10 text-amber-200':'bg-violet-300/10 text-violet-200')}>{group.conflict?'Есть разногласия':'Вес '+group.weight+'/6'}</span>
  </div>
  <div className="flex flex-wrap items-center gap-1.5 rounded-lg border border-white/10 bg-[#121a2e]/70 px-2 py-2">
   <span className="mr-1 text-[10px] text-slate-400">Guardian</span>
   {passport.evidence.map(e=><span key={e.frame} title={e.label} className={'rounded-md border px-1.5 py-1 text-[10px] '+(e.status==='supports'?'text-emerald-200 border-emerald-500/25 bg-emerald-500/5':e.status==='conflicts'?'text-rose-200 border-rose-500/30 bg-rose-500/5':'text-slate-400 border-slate-600/30')}>{e.frame.toUpperCase()} {e.status==='supports'?'✓':e.status==='conflicts'?'!':'·'}</span>)}
   <span className="ml-auto text-[10px] text-slate-400">{passport.supported} за · {passport.conflicted} против</span>
  </div>
  <ScenarioCard scenario={lead} now={now} onDemo={onDemo} isSaved={savedIds?.has(lead.id)??false}/>
  {group.scenarios.length>1&&<button type="button" aria-expanded={expanded} onClick={()=>setExpanded(v=>!v)} className="min-h-[44px] w-full rounded-lg border border-violet-400/25 bg-violet-500/10 px-3 text-xs font-semibold text-violet-200">{expanded?'Скрыть дополнительные таймфреймы':"Другие таймфреймы ("+(group.scenarios.length-1)+")"}</button>}
  {expanded&&group.scenarios.slice(1).map(s=><ScenarioCard key={s.id} scenario={s} now={now} onDemo={onDemo} isSaved={savedIds?.has(s.id)??false}/>)}
 </section>;
}
