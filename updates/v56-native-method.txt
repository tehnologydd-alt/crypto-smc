    /** APEX 5.6: inspect the one existing periodic WorkManager job off the UI thread.
     * We NEVER mistake Android Doze / network delays for a missing schedule.
     * Only repair when opt-in CORE is enabled and no active job actually exists.
     */
    @PluginMethod public void coreReliability(PluginCall call){
        new Thread(() -> {
            try{
                SharedPreferences p=prefs();
                long checked=System.currentTimeMillis();
                boolean coreEnabled=p.getBoolean("coreEnabled",false);
                boolean personal=p.getBoolean("enabled",false);
                java.util.List<androidx.work.WorkInfo> jobs=WorkManager.getInstance(getContext())
                    .getWorkInfosForUniqueWork(JOB).get(6,TimeUnit.SECONDS);
                boolean scheduled=false;
                String state="MISSING";
                for(androidx.work.WorkInfo w:jobs){
                    androidx.work.WorkInfo.State ws=w.getState();
                    if(ws==androidx.work.WorkInfo.State.ENQUEUED||
                       ws==androidx.work.WorkInfo.State.RUNNING||
                       ws==androidx.work.WorkInfo.State.BLOCKED){
                        scheduled=true;
                        state=ws.name();
                        break;
                    }
                }
                boolean repaired=false;
                long lastRepair=p.getLong("coreScheduleRepairAt",0);
                if(coreEnabled&&!scheduled&&(lastRepair<=0||checked-lastRepair>=30*60*1000L)){
                    p.edit().putLong("coreScheduleRepairAt",checked).apply();
                    // Existing shared scheduler observes both CORE and personal preferences.
                    adjustSharedSchedule();
                    repaired=true;
                    state="REQUEUED";
                }
                if(!coreEnabled&&!personal&&!scheduled)state="DISABLED";
                int count=0;
                try{count=new JSONArray(p.getString("coreHistory","[]")).length();}
                catch(Exception ignored){}
                JSObject out=new JSObject();
                out.put("supported",true);
                out.put("enabled",coreEnabled);
                out.put("scheduled",scheduled||repaired);
                out.put("workState",state);
                out.put("repairQueued",repaired);
                out.put("checkedAt",checked);
                out.put("lastRepairAt",p.getLong("coreScheduleRepairAt",0));
                out.put("lastRunAt",p.getLong("coreLastRunAt",0));
                out.put("bufferEntries",count);
                out.put("lastResult",p.getString("coreLastResult","Наблюдений ещё нет"));
                call.resolve(out);
            }catch(InterruptedException e){
                Thread.currentThread().interrupt();
                call.reject("Диагностика CORE прервана",e);
            }catch(Exception e){
                call.reject("Не удалось проверить расписание CORE",e);
            }
        },"apex-core-reliability").start();
    }
