    @PluginMethod public void saveResearchBackup(PluginCall call){
        String content=call.getString("content","");
        String filename=call.getString("filename","APEX-Research.json");
        if(content==null||content.length()<20||content.length()>4000000||!content.contains("apex-research-vault")){call.reject("Неверный архив исследований");return;}
        if(filename==null||!filename.matches("APEX-Research-[A-Za-z0-9-]{0,50}\\.json"))filename="APEX-Research.json";
        Intent intent=new Intent(Intent.ACTION_CREATE_DOCUMENT);
        intent.addCategory(Intent.CATEGORY_OPENABLE);
        intent.setType("application/json");
        intent.putExtra(Intent.EXTRA_TITLE,filename);
        startActivityForResult(call,intent,"writeResearchResult");
    }
    @ActivityCallback private void writeResearchResult(PluginCall call,ActivityResult result){
        if(call==null)return;
        if(result.getResultCode()!=Activity.RESULT_OK||result.getData()==null){call.reject("cancelled");return;}
        Uri uri=result.getData().getData();
        if(uri==null){call.reject("Не выбран файл");return;}
        String contents=call.getString("content","");
        try(OutputStream out=getContext().getContentResolver().openOutputStream(uri,"wt")){
            if(out==null){call.reject("Не удалось открыть файл для записи");return;}
            out.write(contents.getBytes(StandardCharsets.UTF_8));out.flush();
            JSObject answer=new JSObject();answer.put("saved",true);call.resolve(answer);
        }catch(Exception e){call.reject("Не удалось сохранить резервную копию",e);}
    }
    @PluginMethod public void openResearchBackup(PluginCall call){
        Intent intent=new Intent(Intent.ACTION_OPEN_DOCUMENT);
        intent.addCategory(Intent.CATEGORY_OPENABLE);
        intent.setType("application/json");
        startActivityForResult(call,intent,"readResearchResult");
    }
    @ActivityCallback private void readResearchResult(PluginCall call,ActivityResult result){
        if(call==null)return;
        if(result.getResultCode()!=Activity.RESULT_OK||result.getData()==null){call.reject("cancelled");return;}
        Uri uri=result.getData().getData();
        if(uri==null){call.reject("Не выбран файл");return;}
        try(InputStream input=getContext().getContentResolver().openInputStream(uri)){
            if(input==null){call.reject("Не удалось открыть файл");return;}
            InputStreamReader reader=new InputStreamReader(input,StandardCharsets.UTF_8);
            StringBuilder contents=new StringBuilder();char[] buffer=new char[8192];int count;
            while((count=reader.read(buffer))!=-1){
                contents.append(buffer,0,count);
                if(contents.length()>4000000){call.reject("Файл превышает 4 МБ");return;}
            }
            JSObject answer=new JSObject();answer.put("content",contents.toString());call.resolve(answer);
        }catch(Exception e){call.reject("Не удалось прочитать архив",e);}
    }
