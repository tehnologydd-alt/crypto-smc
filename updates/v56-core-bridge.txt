/** Native WorkManager facts are requested on app opening/resume, not assumed.
 * The native method alone can repair a truly missing, opt-in schedule.
 */
export type CoreReliabilityNativeState={
 supported:boolean;enabled:boolean;scheduled:boolean;
 workState:'ENQUEUED'|'RUNNING'|'BLOCKED'|'REQUEUED'|'MISSING'|'DISABLED'|'UNKNOWN';
 repairQueued:boolean;lastRepairAt:number;checkedAt:number;lastRunAt:number;bufferEntries:number;lastResult:string;
};
export async function getCoreReliabilityStatus():Promise<CoreReliabilityNativeState>{
 if(!Capacitor.isNativePlatform())return {
  supported:false,enabled:false,scheduled:false,workState:'UNKNOWN',
  repairQueued:false,lastRepairAt:0,checkedAt:0,lastRunAt:0,bufferEntries:0,lastResult:''
 };
 const raw=await plugin.coreReliability();
 if(!raw||typeof raw!=='object')throw Error('Android CORE не вернул диагностику');
 const x=raw as Record<string,unknown>;
 const valid=['ENQUEUED','RUNNING','BLOCKED','REQUEUED','MISSING','DISABLED','UNKNOWN'];
 return {supported:true,enabled:x.enabled===true,scheduled:x.scheduled===true,
  workState:(typeof x.workState==='string'&&valid.includes(x.workState)?x.workState:'UNKNOWN') as CoreReliabilityNativeState['workState'],
  repairQueued:x.repairQueued===true,
  lastRepairAt:finite(x.lastRepairAt)&&x.lastRepairAt>=0?x.lastRepairAt:0,
  checkedAt:finite(x.checkedAt)&&x.checkedAt>0?x.checkedAt:0,
  lastRunAt:finite(x.lastRunAt)&&x.lastRunAt>=0?x.lastRunAt:0,
  bufferEntries:finite(x.bufferEntries)?Math.min(96,Math.max(0,Math.trunc(x.bufferEntries))):0,
  lastResult:typeof x.lastResult==='string'?x.lastResult.slice(0,220):''};
}
